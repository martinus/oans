# ---------------------------------------------------------------- #273

# a buffer that reaches a preallocated extent is faked whole, data and all
# The upstream shape: data extents are walked past, and the first preallocated
# one decides for the whole buffer. Two files with different data before the
# same preallocated tail then get one digest.
<<<
		if (!(e->fe_flags & FIEMAP_SKIP_FLAGS))
			return false;
===
		if (!(e->fe_flags & FIEMAP_SKIP_FLAGS)) {
			start = e->fe_logical + e->fe_length;
			continue;
		}
>>>

# on a hole, an extent past the end of the area decides for it
# get_extent() answers the *next* extent for an offset in a hole, which can
# start megabytes after the buffer ends.
<<<
		if (!e || e->fe_logical >= end)
===
		if (!e)
>>>

# ---------------------------------------------------------------- #278

# a file that became a directory aborts the run again
# The walk queued a regular file; by the time the consumer gets it, it is a
# directory. Aborting loses the open write batch - up to ten seconds of hashes.
<<<
	if (!S_ISREG(st->stx_mode)) {
		filescan_count_skip(SCAN_SKIP_NOT_REGULAR);
		return 0;
	}
===
	abort_on(!S_ISREG(st->stx_mode));
>>>
